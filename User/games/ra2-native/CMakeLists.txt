cmake_minimum_required(VERSION 3.20)
project(ra2_ports C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(option SHARED TEST TESTS X11 WAYLAND KMSDRM RPI DIRECTFB VIVANTE OPENGL OPENGLES VULKAN ALSA JACK PULSEAUDIO PIPEWIRE SNDIO OSS ESD NAS ARTS HIDAPI JOYSTICK HAPTIC SENSOR UDEV DBUS IBUS FCITX LIBSAMPLERATE LIBUDEV OFFSCREEN)
  set(SDL_${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_UNIX_CONSOLE_BUILD ON CACHE BOOL "" FORCE)
set(SDL_LIBC ON CACHE BOOL "" FORCE)
set(SDL_PTHREADS ON CACHE BOOL "" FORCE)
add_subdirectory("${SDL_DIR}" sdl EXCLUDE_FROM_ALL)
add_library(lixin_sdl STATIC video.c emutls.c)
target_include_directories(lixin_sdl PRIVATE "${SDL_DIR}/src" "${CMAKE_CURRENT_SOURCE_DIR}/../../gui")
target_link_libraries(lixin_sdl PUBLIC SDL2::SDL2-static)
function(lixin_display target)
  target_link_libraries(${target} PRIVATE lixin_sdl)
  target_link_options(${target} PRIVATE "-Wl,--wrap=__emutls_get_address")
  foreach(symbol SDL_Init SDL_InitSubSystem SDL_CreateWindow SDL_CreateRenderer SDL_RenderPresent SDL_UpdateWindowSurface SDL_PumpEvents SDL_PollEvent)
    target_link_options(${target} PRIVATE "-Wl,--wrap=${symbol}")
  endforeach()
endfunction()
include("${ENGINE_DIR}/cmake/ra2_sources.cmake")
add_executable(ra2-engine ${RA2_ENGINE_SOURCES} ${RA2_LIB_SOURCES} "${RA2_ROOT}/engine/audio_sdl.cpp")
target_include_directories(ra2-engine PRIVATE "${RA2_ROOT}/include" "${RA2_ROOT}/engine" "${RA2_ROOT}/third_party" "${RA2_GENERATED_DIR}")
target_compile_definitions(ra2-engine PRIVATE RA2_VERSION="${RA2_VERSION}")
target_compile_options(ra2-engine PRIVATE -ffp-contract=off)
lixin_display(ra2-engine)
set_source_files_properties("${RA2_ROOT}/engine/mp/mp_socket.cpp" PROPERTIES COMPILE_OPTIONS "-include;sys/select.h")
target_sources(ra2-engine PRIVATE engine-path.c)
target_link_options(ra2-engine PRIVATE -Wl,--wrap=main)
set_property(TARGET ra2-engine APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../cpython/program.ld")
if(RECOMP_DIR)
  include(recomp/targets.cmake)
endif()
